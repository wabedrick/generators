  <div class="col-lg-12 chart-container">
        <div class="card">
          <div class="card-body">
            <h5 class="card-title fs-4 text-center">Bar Graph of Sites By their Departments</h5>

            <!-- Bar Chart -->
            <canvas id="barChart" style="max-height: 400px;"></canvas>

            <script>
              document.addEventListener("DOMContentLoaded", () => {
                const departments = <?php echo json_encode($departments); ?>; // PHP array to JS
                const siteCounts = <?php echo json_encode($site_counts); ?>; // PHP array to JS

                const barChart = new Chart(document.querySelector('#barChart'), {
                  type: 'bar',
                  data: {
                    labels: departments, // Departments as labels
                    datasets: [{
                      label: 'Number of Sites',
                      data: siteCounts, // Count of sites per department
                      backgroundColor: 'rgba(75, 192, 192, 0.2)',
                      borderColor: 'rgba(75, 192, 192, 1)',
                      borderWidth: 1
                    }]
                  },
                  options: {
                    scales: {
                      y: {
                        beginAtZero: true
                      }
                    },
                    onClick: function(evt, activeElements) {
                      // Check if a bar was clicked
                      if (activeElements.length > 0) {
                        const index = activeElements[0].index; // Get the clicked bar index
                        const clickedDepartment = departments[index]; // Get the department label

                        // Redirect to the details page with the department name as a parameter
                        window.location.href = `department-details.php?department=${clickedDepartment}`;
                      }
                    },
                    onHover: function(evt, activeElements) {
                      // Change cursor to pointer if hovering over a bar
                      const chartCanvas = evt.native.target;
                      chartCanvas.style.cursor = activeElements.length > 0 ? 'pointer' : 'default';
                    }
                  }
                });
              });
            </script>
            <!-- End Bar Chart -->
          </div>
        </div>
      </div>
